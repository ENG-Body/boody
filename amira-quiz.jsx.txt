export default function App() {
  const [current, setCurrent] = useState(0);
  const [selected, setSelected] = useState(null);
  const [answered, setAnswered] = useState(false);
  const [showFinal, setShowFinal] = useState(false);

  const handleAnswer = (opt) => {
    if (answered) return;
    setSelected(opt);
    setAnswered(true);
  };

  const handleNext = () => {
    if (current + 1 >= questions.length) {
      setShowFinal(true);
    } else {
      setCurrent((c) => c + 1);
      setSelected(null);
      setAnswered(false);
    }
  };

  const progress = (current / questions.length) * 100;
  const q = questions[current];

  if (showFinal) {
    return (
      <>
        <style>{CSS}</style>
        <div className="final-page">
          {HEARTS_DATA.map((h) => (
            <span
              key={h.id}
              className="heart-float"
              style={{
                left: `${h.left}%`,
                bottom: `-${h.bottom}px`,
                fontSize: `${h.size}px`,
                animationDuration: `${h.duration}s`,
                animationDelay: `${h.delay}s`,
              }}
            >
              {h.emoji}
            </span>
          ))}

          <div className="final-card">
            <div className="final-stripe" />
            <span className="big-heart">💝</span>
            <h1 className="final-title">النتيجة النهائية</h1>

            <div className="wrong-box">
              <p className="wrong-box-title">❌ كل الإجابات كانت غلط</p>
              <p className="wrong-box-sub">
                الدنيا دي كلها فيها إجابة صح واحدة بس...
                <br />
                وهي كانت قدامك دايماً
              </p>
            </div>

            <div className="reveal-box">
              <p className="reveal-label">✨ الإجابة الصح الوحيدة في الكون كله هي...</p>
              <p className="reveal-name">أميرة ❤️</p>
            </div>

            <p className="final-note">
              مفيش إجابة صح في الدنيا دي...
              <br />
              <strong>غير اسمك أنتِ 🌹</strong>
              <br />
              لأنكِ أنتِ الحاجة الصح الوحيدة في الكون ده كله 🌟
            </p>
          </div>
        </div>
      </>
    );
  }

  return (
    <>
      <style>{CSS}</style>
      <div className="page" key={current}>
        <div className="card">
          <div className="card-stripe" />

          <p className="q-meta">
            سؤال {current + 1} من {questions.length}
          </p>

          <div className="progress-track">
            <div className="progress-fill" style={{ width: `${progress}%` }} />
          </div>

          <h2 className="question">{q.q}</h2>

          <div className="options">
            {q.options.map((opt, i) => {
              const isSelected = answered && selected === opt;
              const isCorrect = isSelected && opt === "أميرة";
              const isWrong   = isSelected && opt !== "أميرة";
              return (
                <button
                  key={i}
                  onClick={() => handleAnswer(opt)}
                  disabled={answered}
                  className={`opt-btn${isCorrect ? " correct-pick" : isWrong ? " wrong-pick" : ""}`}
                >
                  <span className={`badge${isCorrect ? " correct" : isWrong ? " wrong" : ""}`}>
                    {LABELS[i]}
                  </span>
                  <span className="opt-text">{opt}</span>
                  {isCorrect && <span style={{ fontSize: "20px" }}>✅</span>}
                  {isWrong   && <span style={{ fontSize: "20px" }}>❌</span>}
                </button>
              );
            })}
          </div>

          {answered && (
            <div className="feedback">
              {selected === "أميرة" ? (
                <p className="correct-label">✅ صح!</p>
              ) : (
                <p className="wrong-label">{wrongMessages[current]}</p>
              )}
              <button className="next-btn" onClick={handleNext}>
                {current + 1 >= questions.length
                  ? "🎁 اعرف النتيجة"
                  : "السؤال التالي ←"}
              </button>
            </div>
          )}
        </div>
      </div>
    </>
  );
}
